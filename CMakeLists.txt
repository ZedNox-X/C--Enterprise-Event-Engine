cmake_minimum_required(VERSION 3.24)
project(cpp_enterprise_event_engine VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable ASan/UBSan" OFF)
if(ENABLE_SANITIZERS)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(event_engine src/core/event.cpp src/concurrency/thread_pool.cpp src/rules/rule_engine.cpp src/rules/cpu_rule.cpp src/observability/metrics.cpp)
target_include_directories(event_engine PUBLIC include)
target_compile_options(event_engine PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(event-engine src/main.cpp)
target_link_libraries(event-engine PRIVATE event_engine)
enable_testing()
find_package(GTest QUIET)
if(GTest_FOUND)
 add_executable(event-tests tests/test_event.cpp tests/test_queue.cpp tests/test_rules.cpp)
 target_link_libraries(event-tests PRIVATE event_engine GTest::gtest_main)
 include(GoogleTest)
 gtest_discover_tests(event-tests)
endif()
